getgenv().RemoveHead = false
getgenv().RemoveAccessories = false
getgenv().FakeLeg = false

repeat task.wait() until game:IsLoaded()

local Players = game:GetService("Players")
local player = Players.LocalPlayer
local StarterGui = game:GetService("StarterGui")
local UserInputService = game:GetService("UserInputService")

local LEG_MESH_ID = "101851696"
local LEG_TEX_ID = "101851254"

local headState = false
local accState = false
local legState = false

local function hideHead(character)
    if not character then return end
    local head = character:FindFirstChild("Head")
    if head then
        head.Transparency = 1
        for _, child in ipairs(head:GetChildren()) do
            if child:IsA("ParticleEmitter") then
                child.Enabled = false
            elseif child:IsA("Decal") then
                child.Transparency = 1
            end
        end
    end
    local face = character:FindFirstChild("Face")
    if face then
        face.Transparency = 1
    end
end

local function showHead(character)
    if not character then return end
    local head = character:FindFirstChild("Head")
    if head then
        head.Transparency = 0
        for _, child in ipairs(head:GetChildren()) do
            if child:IsA("ParticleEmitter") then
                child.Enabled = true
            elseif child:IsA("Decal") then
                child.Transparency = 0
            end
        end
    end
    local face = character:FindFirstChild("Face")
    if face then
        face.Transparency = 0
    end
end

local function hideAccessories(character)
    if not character then return end
    for _, obj in ipairs(character:GetDescendants()) do
        if obj:IsA("Accessory") then
            local handle = obj:FindFirstChild("Handle")
            if handle and handle:IsA("BasePart") then
                handle.Transparency = 1
            end
        elseif obj:IsA("Hat") or obj:IsA("Tool") then
            local handle = obj:FindFirstChild("Handle")
            if handle and handle:IsA("BasePart") then
                handle.Transparency = 1
            end
        end
    end
end

local function showAccessories(character)
    if not character then return end
    for _, obj in ipairs(character:GetDescendants()) do
        if obj:IsA("Accessory") then
            local handle = obj:FindFirstChild("Handle")
            if handle and handle:IsA("BasePart") then
                handle.Transparency = 0
            end
        elseif obj:IsA("Hat") or obj:IsA("Tool") then
            local handle = obj:FindFirstChild("Handle")
            if handle and handle:IsA("BasePart") then
                handle.Transparency = 0
            end
        end
    end
end

local function applyLegEffect(character)
    if not character then return end
    local leg = character:FindFirstChild("Right Leg") or character:FindFirstChild("RightLowerLeg")
    if not leg then return end

    for _, child in ipairs(character:GetChildren()) do
        if child.Name == "__FakeLegPart" then
            child:Destroy()
        end
    end

    local fakeLeg = Instance.new("Part")
    fakeLeg.Name = "__FakeLegPart"
    fakeLeg.Size = leg.Size
    fakeLeg.CFrame = leg.CFrame
    fakeLeg.Anchored = false
    fakeLeg.CanCollide = false
    fakeLeg.Transparency = 0
    fakeLeg.BrickColor = BrickColor.new("Really black")
    fakeLeg.Material = Enum.Material.Plastic

    local mesh = Instance.new("SpecialMesh")
    mesh.MeshType = Enum.MeshType.FileMesh
    mesh.MeshId = "rbxassetid://" .. LEG_MESH_ID
    mesh.TextureId = "rbxassetid://" .. LEG_TEX_ID
    mesh.Scale = Vector3.new(1, 1, 1)
    mesh.Parent = fakeLeg

    local weld = Instance.new("WeldConstraint")
    weld.Part0 = leg
    weld.Part1 = fakeLeg
    weld.Parent = fakeLeg

    fakeLeg.Parent = character
    leg.Transparency = 1
end

local function removeLegEffect(character)
    for _, child in ipairs(character:GetChildren()) do
        if child.Name == "__FakeLegPart" then
            child:Destroy()
        end
    end
    local leg = character:FindFirstChild("Right Leg") or character:FindFirstChild("RightLowerLeg")
    if leg then
        leg.Transparency = 0
    end
end

local function notify(title, text, duration)
    pcall(function()
        StarterGui:SetCore("SendNotification", {
            Title = title,
            Text = text,
            Duration = duration or 5
        })
    end)
end

local function setupGUI()
    local screenGui = Instance.new("ScreenGui")
    screenGui.Name = "EffectToggleGUI"
    screenGui.ResetOnSpawn = false
    screenGui.Parent = player:WaitForChild("PlayerGui")

    local frame = Instance.new("Frame")
    frame.Size = UDim2.new(0, 200, 0, 180)
    frame.Position = UDim2.new(0, 500, 0, 10)
    frame.BackgroundColor3 = Color3.new(0.2, 0.2, 0.2)
    frame.BackgroundTransparency = 0.3
    frame.BorderSizePixel = 2
    frame.BorderColor3 = Color3.new(1, 1, 1)
    frame.Draggable = true
    frame.Parent = screenGui

    local title = Instance.new("TextLabel")
    title.Size = UDim2.new(1, 0, 0, 25)
    title.BackgroundTransparency = 1
    title.Text = "效果切换"
    title.TextColor3 = Color3.new(1, 1, 1)
    title.TextScaled = true
    title.Font = Enum.Font.SourceSansBold
    title.Parent = frame

    local btnHead = Instance.new("TextButton")
    btnHead.Size = UDim2.new(0.9, 0, 0, 30)
    btnHead.Position = UDim2.new(0.05, 0, 0, 30)
    btnHead.BackgroundColor3 = Color3.new(0.3, 0.6, 1)
    btnHead.Text = "头部透明 (关)"
    btnHead.TextColor3 = Color3.new(1, 1, 1)
    btnHead.Parent = frame

    local btnAcc = Instance.new("TextButton")
    btnAcc.Size = UDim2.new(0.9, 0, 0, 30)
    btnAcc.Position = UDim2.new(0.05, 0, 0, 65)
    btnAcc.BackgroundColor3 = Color3.new(0.3, 0.6, 1)
    btnAcc.Text = "饰品隐藏 (关)"
    btnAcc.TextColor3 = Color3.new(1, 1, 1)
    btnAcc.Parent = frame

    local btnLeg = Instance.new("TextButton")
    btnLeg.Size = UDim2.new(0.9, 0, 0, 30)
    btnLeg.Position = UDim2.new(0.05, 0, 0, 100)
    btnLeg.BackgroundColor3 = Color3.new(0.3, 0.6, 1)
    btnLeg.Text = "虚假腿部 (关)"
    btnLeg.TextColor3 = Color3.new(1, 1, 1)
    btnLeg.Parent = frame

    local btnAll = Instance.new("TextButton")
    btnAll.Size = UDim2.new(0.9, 0, 0, 30)
    btnAll.Position = UDim2.new(0.05, 0, 0, 135)
    btnAll.BackgroundColor3 = Color3.new(0.8, 0.3, 0.3)
    btnAll.Text = "一键开关"
    btnAll.TextColor3 = Color3.new(1, 1, 1)
    btnAll.Parent = frame

    local function updateButtons()
        btnHead.Text = "头部透明 " .. (headState and "(开)" or "(关)")
        btnAcc.Text = "饰品隐藏 " .. (accState and "(开)" or "(关)")
        btnLeg.Text = "虚假腿部 " .. (legState and "(开)" or "(关)")
    end

    local function applyAllStates(character)
        if headState then hideHead(character) else showHead(character) end
        if accState then hideAccessories(character) else showAccessories(character) end
        if legState then applyLegEffect(character) else removeLegEffect(character) end
    end

    local function toggleAll()
        local char = player.Character
        if not char then return end
        local allOn = headState and accState and legState
        if allOn then
            headState = false
            accState = false
            legState = false
            showHead(char)
            showAccessories(char)
            removeLegEffect(char)
        else
            headState = true
            accState = true
            legState = true
            hideHead(char)
            hideAccessories(char)
            applyLegEffect(char)
        end
        updateButtons()
    end

    btnHead.MouseButton1Click:Connect(function()
        local char = player.Character
        if not char then return end
        if headState then
            showHead(char)
            headState = false
        else
            hideHead(char)
            headState = true
        end
        updateButtons()
    end)

    btnAcc.MouseButton1Click:Connect(function()
        local char = player.Character
        if not char then return end
        if accState then
            showAccessories(char)
            accState = false
        else
            hideAccessories(char)
            accState = true
        end
        updateButtons()
    end)

    btnLeg.MouseButton1Click:Connect(function()
        local char = player.Character
        if not char then return end
        if legState then
            removeLegEffect(char)
            legState = false
        else
            applyLegEffect(char)
            legState = true
        end
        updateButtons()
    end)

    btnAll.MouseButton1Click:Connect(toggleAll)

    player.CharacterAdded:Connect(function(character)
        task.wait(0.3)
        applyAllStates(character)
        updateButtons()
    end)

    if player.Character then
        task.defer(function()
            if getgenv().RemoveHead then
                hideHead(player.Character)
                headState = true
            end
            if getgenv().RemoveAccessories then
                hideAccessories(player.Character)
                accState = true
            end
            if getgenv().FakeLeg then
                applyLegEffect(player.Character)
                legState = true
            end
            updateButtons()
        end)
    end

    updateButtons()

    UserInputService.InputBegan:Connect(function(input, gp)
        if gp then return end
        if input.KeyCode == Enum.KeyCode.T then
            screenGui.Enabled = not screenGui.Enabled
        elseif input.KeyCode == Enum.KeyCode.R then
            toggleAll()
        end
    end)

    return screenGui
end

setupGUI()
notify("fake headless and korblox", "作者yuela，T键开关UI，R键一键开关所有效果。Q群1005801189，有问题请反馈", 5)
